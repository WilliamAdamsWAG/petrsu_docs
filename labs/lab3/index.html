void show_lychrel_candidates(long last_number)
{

    for (number = 1; number <= last_number; number++) {
        if (is_lychrel_candidate(number)) {
            printf("%ld\n", number);
        }

int is_lychrel_candidate(long number)
{
    long n = number;
    long r = reverse(n);

    do {
        if (n > LONG_MAX - r) {
            return 1;
        }
        n = n + r;

        r = reverse(n);
    } while (n != r);

    return 0;
}

long reverse(long n)
{
    long r = 0;

    do {
        r = r * 10 + n % 10;
        n /= 10;
    } while (n > 0);

    return r;
}

    Объяснение кода

    Принцип работы

    Сама программа работает по простому принципу: 1. Пользователь вводит верхнюю границу, число до которого мы ищем нужные числа 2. Программа перебирает все числа от 1 до верхней границы, если число является кандидатом в числа Лишрел то выводим его в консоль

    Данная функция принимает на входе переменную last_number, это верхняя граница диапазона рассматриваемых чисел. Далее в цикле мы проходим по всем значениям и проверяем каждое число с помощью функции is_lychrel_candidate(), если число подходит - печатаем его.

    На входе функция принимает рассматриваемое число. У нас объявляется 2 новые переменные: n и r. n это рассматриваемое число, r - его перевёрнутая версия.

    Для проверки на кандидата в числа Лишрел мы используем цикл do-while, пока числа n и r не равны проверка продолжается, если равны то искомое число не является кандидатом в числа Лишрел. Внутри цикла предусмотрена проверка на переполнение:

    Переполнение это выход за диапазон допустимых значений, в этом случае например при сложении лишние байты в числе просто отсекут, что исказит значение. Переполнение может произойти вследствие сложения n и r, поэтому составим уравнение для проверки:

    n + r > LONG_MAX

    Однако если мы их сложим и они вызовут переполнение, то из-за искажения условие может выполниться некорректно, поэтому немного изменим условие:

    n > LONG_MAX - r

    Если это условие выполняется значит мы превысили лимит для типа данных long, и следовательно искомое число кандидат в числа Лишрел, поэтому мы возвращаем единицу

    long reverse(long n)
{
    long r = 0;

    do {
        r = r * 10 + n % 10;
        n /= 10;
    } while (n > 0);

    return r;
}

    Функция принимает одно число n, значение которого требуется перевернуть, и возвращает это перевёрнутое число. Для хранения результата объявляется переменная r, изначально равная нулю.

    Далее запускается цикл do-while, который выполняется до тех пор, пока n больше нуля. На каждой итерации мы "добавляем" к результату последнюю цифру числа n:

    • операция n % 10 извлекает последнюю цифру числа,
    • умножение r * 10 сдвигает текущее число влево, освобождая место для новой цифры,
    • к этому результату прибавляется последняя цифра, тем самым формируя перевёрнутое число шаг за шагом.

    После этого n делится на 10 (n /= 10), чтобы "удалить" последнюю цифру, и процесс повторяется. Когда все цифры закончились (n становится 0), цикл завершается, и функция возвращает готовое перевёрнутое число r.

    Нам необходимо доработать функцию reverse() чтобы она при переполнении возвращала -1, для этого добавим следующее:

    long reverse(long n)
{
    long r = 0;

    do {
        if (r > (LONG_MAX - n%10) / 10) {
            return -1;
        }
        r = r * 10 + n % 10;
        n /= 10;
    } while (n > 0);

    return r;

    Почему такая проверка?

    Переполнение может произойти в строке r = r * 10 + n % 10;(т.к. мы r умножаем на 10) поэтому мы решаем следующее уравнение:

    r*10 + n%10 > LONG_MAX

    r*10 > LONG_MAX - n%10

    r > (LONG_MAX - n%10) / 10

    Теперь добавим обработку этого случая внутри функции is_lychrel_candidate():

int is_lychrel_candidate(long number)
{
    long n = number;
    long r = reverse(n);

    if (r == -1) {
        return 0;
    }

    do {
        if (n > LONG_MAX - r) {
            return 1;

        n = n + r;

        r = reverse(n);
    } while (n != r);

    return 0;
}

{

            return -1;

        n /= 10;
